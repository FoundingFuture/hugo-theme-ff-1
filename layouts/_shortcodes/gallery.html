{{- /* Thumbnails that open full size when clicked. One picture a line: the file first, the caption after a pipe. The files are page resources, so they sit in the page's own folder beside its index.md, and Hugo cuts the thumbnails. No script. The full-size view is a :target, so clicking a thumbnail changes the fragment and CSS does the rest. Back closes it, and a link to any picture can be sent to somebody. */ -}} {{- if partial "feature-on.html" (dict "name" "prose" "page" .Page) }} {{- $id := printf "g%d" .Ordinal }} {{- $rows := slice }} {{- range (split (trim .Inner "\n") "\n") }} {{- $cut := split . "|" }} {{- $src := trim (index $cut 0) " " }} {{- $cap := "" }}{{ if ge (len $cut) 2 }}{{ $cap = trim (index $cut 1) " " }}{{ end }} {{- with $src }}{{ $rows = $rows | append (dict "src" . "cap" $cap) }}{{ end }} {{- end }} {{- $n := len $rows }} {{- $fmt := cond hugo.IsExtended "webp" "jpg" }}
{{- range $i, $r := $rows }} {{- with $.Page.Resources.GetMatch $r.src }} {{- /* Fill, not Fit: one tall picture among wide ones makes a ragged grid three rows deep. Anchored top, because the top of a screen is the part that says which screen it is. */}} {{- $thumb := .Fill (printf "560x380 top %s q82" $fmt) }} {{ $r.cap }} {{- end }} {{- end }}
{{- range $i, $r := $rows }} {{- with $.Page.Resources.GetMatch $r.src }} {{- $full := .Fit (printf "1600x1600 %s q86" $fmt) }} {{- $prev := printf "#%s-%d" $id (cond (eq $i 0) (sub $n 1) (sub $i 1)) }} {{- $next := printf "#%s-%d" $id (cond (eq $i (sub $n 1)) 0 (add $i 1)) }}
{{- if gt $n 1 }} ‹ › {{- end }} ×
{{- end }} {{- end }} {{- end }}
