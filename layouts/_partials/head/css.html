{{- /* The theme's own stylesheet, then one per enabled feature, joined into a single file. A feature that is off adds nothing. The switch is read from the site, not from a page. A stylesheet is served to the whole site, so a per-page override cannot change which rules it carries. */ -}} {{- $sheets := slice }} {{- with resources.Get "css/main.css" }} {{- $sheets = $sheets | append . }} {{- end }} {{- /* The code colours, which the theme owns rather than inherits as inline styles it cannot override. */ -}} {{- with resources.Get "css/chroma.css" }} {{- $sheets = $sheets | append . }} {{- end }} {{- /* The rules that keep a link visible and reachable. */ -}} {{- with resources.Get "css/a11y.css" }} {{- $sheets = $sheets | append . }} {{- end }} {{- $features := slice }} {{- range $name, $manifest := index hugo.Data "ff-1" "features" }} {{- $on := $manifest.default }} {{- with index site.Params "ff-1" "features" }} {{- $found := index . $name }} {{- if ne $found nil }}{{ $on = $found }}{{ end }} {{- end }} {{- if and $on $manifest.css }} {{- $features = $features | append (dict "weight" (default 50 $manifest.weight) "css" $manifest.css) }} {{- end }} {{- end }} {{- range sort $features "weight" }} {{- with resources.Get .css }} {{- $sheets = $sheets | append . }} {{- end }} {{- end }} {{- /* The site's own, last of all, so setting a property again changes it. The theme ships no css/custom.css: resources.Get reads the union of the theme's assets and the site's, so this finds a file only where a site has written one. Last is the whole point. A site that cannot put a rule after the theme's cannot override anything, and every custom property this theme documents is unreachable. */ -}} {{- with resources.Get "css/custom.css" }} {{- $sheets = $sheets | append . }} {{- end }} {{- /* The font licences, published beside the site rather than left in assets where nothing copies them. The faces are served, so the terms they travel under have to be served too. The OFL allows the notice to ride inside the binary, and every face here does carry its copyright in its own name table, but a name table is not a thing a reader can open. These two are: the licence, and the notice for each face with what was done to it. Published from here because this is where the faces enter the build. */ -}} {{- range slice "fonts/OFL.txt" "fonts/README.txt" }} {{- with resources.Get . }}{{ .Publish }}{{ end }} {{- end }} {{- with $sheets }} {{- $opts := dict "minify" (cond hugo.IsDevelopment false true) "sourceMap" (cond hugo.IsDevelopment "linked" "none") }} {{- with . | resources.Concat "css/bundle.css" | css.Build $opts }} {{- if hugo.IsDevelopment }} {{- else }} {{- with . | fingerprint }} {{- end }} {{- end }} {{- end }} {{- end }}
