{{- /* Render every enabled feature registered for one slot, in weight order. @context {string} slot The slot name, such as "page.meta". @context {page} page The page being rendered. Every key this theme reads sits under one namespace, which Hugo asks theme authors to do so that a site's own params and a second theme's cannot collide with them. index, not dot notation, because a slug may carry a hyphen and is then not a valid template identifier. A slot no feature registers for is the common case, and it costs one lookup here. @example: {{ partial "slot.html" (dict "slot" "page.meta" "page" .) }} */ -}} {{- $slot := .slot }} {{- $page := .page }} {{- with index (partialCached "slot-map.html" "features") $slot }} {{- $on := partialCached "feature-set.html" $page $page.RelPermalink }} {{- range . }} {{- if and (index $on .name) (or (not .kinds) (in .kinds $page.Kind)) }} {{- partial .partial $page }} {{- end }} {{- end }} {{- end }}
