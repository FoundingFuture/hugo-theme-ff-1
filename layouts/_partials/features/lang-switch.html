{{- /* The languages a site publishes. Nothing at all on a site with one language: hugo.Sites has one entry and there is no choice to offer. A language links to this page in that language when there is one. When there is not, it links to that language's home, because Hugo builds each language as its own site and that home lists what the language has. The alternative is a switch that only works on pages already translated, which is a switch you cannot use to go and look. AllTranslations carries the current page as well, so the language being read resolves to itself and needs no special case. */ -}} {{- $page := . }} {{- if gt (len hugo.Sites) 1 }}
{{ i18n "availableLanguages" }} {{- range $s := hugo.Sites }} {{- $target := $s.Home }} {{- $exact := false }} {{- range $page.AllTranslations }} {{- if eq .Language.Lang $s.Language.Lang }} {{- $target = . }}{{ $exact = true }} {{- end }} {{- end }} {{ $s.Language.Lang | upper }} {{- end }}
{{- end }}
