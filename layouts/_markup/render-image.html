{{- /* A Markdown image, sized for the screen that asks for it. Without responsive-images this renders exactly what Hugo's own image hook renders: the destination resolved against the page's resources, then the site's, and passed through untouched when it is neither. conform compares the result against that rendering, so anything else here would be a difference on every page carrying a picture. The feature is what declares the difference. Resolution goes through PageInner, which is what Hugo's own hooks use. The two differ when a hook fires while an included page is being rendered: Page is the page included, PageInner the one including it. PageInner falls back to Page and always answers. */ -}} {{- $captions := partial "feature-on.html" (dict "name" "figure-captions" "page" .Page) -}} {{- $responsive := partial "feature-on.html" (dict "name" "responsive-images" "page" .Page) -}} {{- $u := urls.Parse .Destination -}} {{- $src := .Destination -}} {{- $resource := false -}} {{- if not $u.IsAbs -}} {{- with or (.PageInner.Resources.Get $u.Path) (resources.Get $u.Path) -}} {{- $resource = . -}} {{- end -}} {{- end -}} {{- /* The source's own address is read where it is used and nowhere else. A resource is published the moment its permalink is read, and reading it here published every original beside the variants made from it: 1.4MB of PNG that no page ever asked for, because a picture that went through the pipeline is served from the resized copies. An SVG is served from this address, so it is read on that path. */ -}} {{- /* Only a picture the pipeline can open. Hugo is asked rather than told. This read the media type and compared its subtype against svg+xml, which is the spelling in image/svg+xml and not the spelling Hugo reports: it says svg. So the guard never matched, every SVG took the raster path, and the first call to .Width ended the build. Hugo's own error named the function to use, and this is it. A GIF is processable and still passed through, because it may be an animation that resizing would flatten to its first frame. */ -}} {{- $raster := false -}} {{- with $resource -}} {{- if and (reflect.IsImageResourceProcessable .) (ne .MediaType.SubType "gif") -}} {{- $raster = true -}} {{- end -}} {{- end -}} {{- $tag := "" -}} {{- if and $responsive $raster -}} {{- /* The reading measure is 78ch, about 765px, and the widest a picture is ever drawn here. Twice that covers a dense screen. A width above the source is skipped rather than upscaled, so a small picture gets one entry and no invented detail. */ -}} {{- $widths := slice 480 768 1530 -}} {{- $sizes := "(max-width: 62rem) 100vw, 765px" -}} {{- /* A source narrower than every width asked for would otherwise get no variant at all, and a small picture would keep the format it arrived in. It gets one entry at its own width. */ -}} {{- $targets := slice -}} {{- range $widths -}} {{- if le . $resource.Width -}}{{- $targets = $targets | append . -}}{{- end -}} {{- end -}} {{- if not $targets -}}{{- $targets = slice $resource.Width -}}{{- end -}} {{- $avif := slice -}} {{- $webp := slice -}} {{- $wide := $resource -}} {{- range $targets -}} {{- $w := . -}} {{- if true -}} {{- with $resource.Resize (printf "%dx avif q55" $w) -}} {{- $avif = $avif | append (printf "%s %dw" .RelPermalink $w) -}} {{- end -}} {{- with $resource.Resize (printf "%dx webp q80" $w) -}} {{- $webp = $webp | append (printf "%s %dw" .RelPermalink $w) -}} {{- $wide = . -}} {{- end -}} {{- end -}} {{- end -}} {{- /* width and height are the source's, so the browser knows the shape before a byte of it arrives and the page stops jumping as pictures land. Ordinal is the picture's place on the page: the first is likely the one a reader is waiting for, so it is not deferred. */ -}} {{- $loading := cond (eq .Ordinal 0) "eager" "lazy" -}} {{- $img := printf "" $wide.RelPermalink (delimit $webp ", ") $sizes (string $resource.Width) (string $resource.Height) $loading .PlainText -}} {{- if $webp -}} {{- $tag = printf "%s%s%s" (cond (gt (len $avif) 0) (printf "" (delimit $avif ", ") $sizes) "") (printf "" (delimit $webp ", ") $sizes) $img -}} {{- end -}} {{- end -}} {{- /* An SVG carries its size in its own markup, so it can still say how tall it is beside how wide. Without that a browser learns the shape only when the file arrives, and the page moves under the reader exactly as it does for a raster with no dimensions. The width and height attributes first, and the viewBox where the file gives none, which is the common case for a drawing exported for the web. Congo reads them the same way. */ -}} {{- $svgW := "" -}} {{- $svgH := "" -}} {{- if and $resource (eq $resource.MediaType.SubType "svg") -}} {{- $svg := $resource.Content -}} {{- range findRESubmatch `]*[[:space:]]width="([0-9.]+)` $svg 1 -}} {{- $svgW = index . 1 -}} {{- end -}} {{- range findRESubmatch `]*[[:space:]]height="([0-9.]+)` $svg 1 -}} {{- $svgH = index . 1 -}} {{- end -}} {{- if or (eq $svgW "") (eq $svgH "") -}} {{- range findRESubmatch `viewBox="[0-9.-]+[[:space:]]+[0-9.-]+[[:space:]]+([0-9.]+)[[:space:]]+([0-9.]+)` $svg 1 -}} {{- $svgW = index . 1 -}} {{- $svgH = index . 2 -}} {{- end -}} {{- end -}} {{- end -}} {{- /* A caption, where there is one and the feature is on. Decided before the plain img is written, because that one carries the title as a tooltip and a figure must not: the words are already under the picture as its caption, and repeating them puts the same sentence in the accessible name and in a tooltip. v0.2.4 wrote the two branches separately and did not have this. */ -}} {{- $captioned := and $captions (ne (string .Title) "") -}} {{- if not $tag -}} {{- /* Here is where the source is served from, so here is where its address is read and the file published. */ -}} {{- with $resource }}{{ $src = .RelPermalink }}{{ end -}} {{- $tag = printf "" $src .PlainText (cond (and (ne $svgW "") (ne $svgH "")) (printf " width=%q height=%q" $svgW $svgH) "") (cond (and (ne (string .Title) "") (not $captioned)) (printf " title=%q" .Title) "") -}} {{- end -}} {{- if $captioned -}} {{- else -}} {{ $tag | safeHTML }} {{- end -}}
